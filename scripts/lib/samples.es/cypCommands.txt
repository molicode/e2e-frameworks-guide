// cypress/support/commands.ts — encapsulá flujos repetitivos una sola vez.
Cypress.Commands.add("login", (user: string, pass: string) => {
  cy.session([user], () => {
    cy.visit("/login");
    cy.get("[data-cy=user]").type(user);
    cy.get("[data-cy=pass]").type(pass);
    cy.get("[data-cy=submit]").click();
  });
});

// Enseñale a TypeScript el nuevo comando (cypress/support/index.d.ts):
declare global {
  namespace Cypress {
    interface Chainable {
      login(user: string, pass: string): void;
    }
  }
}

cy.login("demo", "secret");
