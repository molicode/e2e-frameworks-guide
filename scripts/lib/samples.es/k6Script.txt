// load-test.js — k6: un test de carga como código, con umbrales de pass/fail.
import http from "k6/http";
import { check, sleep } from "k6";

export const options = {
  // Escalá usuarios virtuales (VUs): 0 -> 50 en 30s, mantené, después bajá.
  stages: [
    { duration: "30s", target: 50 },
    { duration: "1m", target: 50 },
    { duration: "10s", target: 0 },
  ],
  // Los umbrales convierten la corrida en quality gate (falla el build si se rompe).
  thresholds: {
    http_req_failed: ["rate<0.01"],    // menos de 1% de errores
    http_req_duration: ["p(95)<500"],  // 95% de los requests bajo 500ms
  },
};

export default function () {
  const res = http.get("https://api.example.com/orders/42");
  check(res, { "status is 200": (r) => r.status === 200 });
  sleep(1); // think time entre requests
}

// Corré:  k6 run load-test.js
