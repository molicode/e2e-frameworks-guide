# locustfile.py — Locust: modelá el comportamiento de un usuario en Python.
from locust import HttpUser, task, between


class ShopUser(HttpUser):
    # Cada usuario simulado espera 1-3s entre acciones (think time).
    wait_time = between(1, 3)

    @task(3)              # peso 3: leé tres veces más seguido que el checkout
    def view_order(self):
        self.client.get("/orders/42")

    @task(1)
    def checkout(self):
        self.client.post("/orders", json={"items": ["book"]})


# Web UI:            locust -f locustfile.py --host https://api.example.com
# Headless (CI):     locust ... --headless -u 100 -r 10 -t 2m
