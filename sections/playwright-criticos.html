API testing (CRUD + auth)

Probás la API directamente: creás (POST → 201), leés (GET → 200) y validás el contrato de la respuesta. Y verificás la autenticación: sin token, 401; nunca un 200 silencioso.

Python 
# CRUD de API con auth — creá, leé, aseverá el contrato. Sin UI.
api = playwright.request.new_context(
    base_url="https://api.example.com",
    extra_http_headers={"Authorization": f"Bearer {token}"},
)

# CREATE
created = api.post("/orders", data={"items": ["book"]})
assert created.status == 201
order_id = created.json()["id"]

# READ + aseverá la forma de la respuesta
res = api.get(f"/orders/{order_id}")
assert res.ok
assert res.json()["status"] == "NEW"

# Sin un token -> 401 (no un 200 silencioso ni un 500)
anon = playwright.request.new_context()
assert anon.get(f"https://api.example.com/orders/{order_id}").status == 401# API CRUD with auth — create, read, assert the contract. No UI.

Cuentas y recibos de pago

La matemática del recibo tiene que cerrar siempre: la suma de los ítems = subtotal, el impuesto exacto y subtotal + impuesto = total. Más los invariantes: nunca negativo, estado consistente con lo pagado.

Python 
# Recibo de pago — la cuenta del dinero SIEMPRE debe cuadrar.
# Trabajá en centavos enteros para evitar errores de punto flotante.
r = api.get("/api/receipts/9087").json()

# La suma de los ítems es igual al subtotal.
items_total = sum(it["cents"] for it in r["items"])
assert items_total == r["subtotalCents"]                  # 9000

# El impuesto y el total son exactos.
assert r["taxCents"] == round(r["subtotalCents"] * 0.21)  # 1890
assert r["totalCents"] == r["subtotalCents"] + r["taxCents"]  # 10890

# Invariantes: nunca negativo, estado consistente con el monto pagado.
assert r["totalCents"] > 0
assert r["status"] == "PAID"# Payment receipt — the money math must ALWAYS add up.
# Work in integer cents to avoid floating-point errors.
r = api.get("/api/receipts/9087").json()

# Sum of line items equals the subtotal.
items_total = sum(it["cents"] for it in r["items"])
assert items_total == r["subtotalCents"]                  # 9000

# Tax and total are exact.
assert r["taxCents"] == round(r["subtotalCents"] * 0.21)  # 1890
assert r["totalCents"] == r["subtotalCents"] + r["taxCents"]  # 10890

# Invariants: never negative, status consistent with the amount paid.
assert r["totalCents"] > 0
assert r["status"] == "PAID"

Validación de documentos legales

Facturas, contratos o comprobantes: validás que estén los campos obligatorios, el formato del número, el estado (firmado) y que la fecha no sea futura.

Python 
# Validación de documento legal (Playwright) — campos requeridos y formato de la factura.
import re
from datetime import date

page.goto("/invoices/INV-2026-0042")
expect(page.locator(".invoice-number")).to_have_text(re.compile(r"^INV-\d{4}-\d{4}$"))
expect(page.locator(".tax-id")).not_to_be_empty()
expect(page.locator(".doc-status")).to_have_text("SIGNED")
issued = page.locator(".issued-date").text_content()
assert date.fromisoformat(issued) <= date.today()  # no futura# Legal document validation (Playwright) — invoice required fields & format.

Seguridad: autorización e IDOR

El control de acceso es crítico: el usuario A no debe poder leer la orden del usuario B con solo cambiar el id en la URL (eso es un IDOR). Debe responder 403, no 200.

Python 
# Seguridad (Playwright) — autorización/IDOR + XSS.
# El usuario A NO debe leer la factura del usuario B cambiando el id.
as_user_a = playwright.request.new_context(
    extra_http_headers={"Authorization": f"Bearer {user_a_token}"},
)
res = as_user_a.get("/api/invoices/INV-2026-0099")
assert res.status == 403  # prohibido, NO 200

# XSS: el payload se renderiza como texto, nunca se ejecuta.
page.goto("/search?q=<script>alert(1)</script>")
expect(page.get_by_text("<script>alert(1)</script>")).to_be_visible()
expect(page.locator(".results")).to_be_visible()# Security (Playwright) — authorization/IDOR + XSS.
# User A must NOT read user B's invoice by changing the id.
as_user_a = playwright.request.new_context(
    extra_http_headers={"Authorization": f"Bearer {user_a_token}"},
)
res = as_user_a.get("/api/invoices/INV-2026-0099")
assert res.status == 403  # forbidden, NOT 200

# XSS: the payload renders as text, never executes.
page.goto("/search?q=<script>alert(1)</script>")
expect(page.get_by_text("<script>alert(1)</script>")).to_be_visible()
expect(page.locator(".results")).to_be_visible()
