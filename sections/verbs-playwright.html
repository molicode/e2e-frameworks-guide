Verbos HTTP

Con Playwright

Playwright tiene un request context independiente del navegador, con un método por verbo (get, post, put, patch, delete, head) y fetch genérico para el resto.

La respuesta expone .status, .json(), .ok y .headers. Para OPTIONS usás fetch(url, method="OPTIONS") porque no hay un atajo dedicado.

Python 
# Todos los verbos HTTP con el request context de Playwright.
api = playwright.request.new_context(
    base_url="https://api.example.com",
    extra_http_headers={"Authorization": f"Bearer {token}"},
)

# GET — leer (seguro, idempotente)
assert api.get("/orders/42").status == 200

# POST — crear (NO idempotente)
created = api.post("/orders", data={"items": ["book"]})
assert created.status == 201
order_id = created.json()["id"]

# PUT — reemplazar el recurso entero (idempotente)
assert api.put(f"/orders/{order_id}",
               data={"items": ["pen"], "status": "NEW"}).status == 200

# PATCH — actualización parcial
assert api.patch(f"/orders/{order_id}", data={"status": "PAID"}).status == 200

# DELETE — eliminar (idempotente)
assert api.delete(f"/orders/{order_id}").status == 204

# HEAD — solo headers, sin body
assert api.head("/orders/42").status == 200

# OPTIONS — métodos permitidos (usá el fetch genérico)
opt = api.fetch("/orders", method="OPTIONS")
assert "POST" in opt.headers.get("allow", "")# Every HTTP verb with Playwright's request context.
api = playwright.request.new_context(
    base_url="https://api.example.com",
    extra_http_headers={"Authorization": f"Bearer {token}"},
)

# GET — read (safe, idempotent)
assert api.get("/orders/42").status == 200

# POST — create (NOT idempotent)
created = api.post("/orders", data={"items": ["book"]})
assert created.status == 201
order_id = created.json()["id"]

# PUT — replace the whole resource (idempotent)
assert api.put(f"/orders/{order_id}",
               data={"items": ["pen"], "status": "NEW"}).status == 200

# PATCH — partial update
assert api.patch(f"/orders/{order_id}", data={"status": "PAID"}).status == 200

# DELETE — remove (idempotent)
assert api.delete(f"/orders/{order_id}").status == 204

# HEAD — headers only, no body
assert api.head("/orders/42").status == 200

# OPTIONS — allowed methods (use the generic fetch)
opt = api.fetch("/orders", method="OPTIONS")
assert "POST" in opt.headers.get("allow", "")
